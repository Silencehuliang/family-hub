  expires_at  INTEGER NOT NULL,

-- 设备（信任）

  id           TEXT PRIMARY KEY,
  member_id    TEXT NOT NULL REFERENCES sys_member(id),

  last_ip      TEXT,

  trusted      INTEGER NOT NULL DEFAULT 1, -- 0/1

  expires_at  INTEGER NOT NULL,
  created_at  INTEGER NOT NULL DEFAULT (strftime('%s','now'))
);
CREATE INDEX idx_session_member ON sys_session(member_id);

-- 通知
CREATE TABLE sys_notification (
  id          TEXT PRIMARY KEY,
  family_id   TEXT NOT NULL,
  member_id   TEXT,                        -- 接收者；NULL = 全家

-- 通知偏好（每成员每类型）

-- 飞书配置（每家庭一条）

-- Web Push 订阅

```ts
// workers/api/src/modules/auth/AuthService.ts
async login(fingerprint: string, pin: string) {
  const device = await db.prepare(
    'SELECT * FROM sys_device WHERE fingerprint=? AND trusted=1'
  ).bind(fingerprint).first();
  if (!device) throw new BizError('DEVICE_UNTRUSTED');

  const member = await db.prepare('SELECT * FROM sys_member WHERE id=?')
    .bind(device.member_id).first();
  if (member.pin_locked_until > now) throw new BizError('PIN_LOCKED');

  if (!verifyPin(pin, member.pin_hash)) {
    await incPinFail(member.id);          // KV 计数；满 5 → pin_locked_until = now+30min
    throw new BizError('VALIDATION', 'PIN 错误');
  }
  await clearPinFail(member.id);
  const token = await createSession(member.id, device.id);
  return { token, memberId: member.id };
}
```

### 4.5 限流（KV）
- key: `pinfail:<memberId>`，TTL 600s，INCR；达 5 → 写 `pin_locked_until`。

## 5. 账单模块详细设计

### 5.1 列表查询（带筛选 + 标签）
```sql
SELECT b.*, GROUP_CONCAT(t.name) AS tag_names
FROM bill_record b
LEFT JOIN bill_record_tag rt ON rt.record_id = b.id
LEFT JOIN bill_tag t ON t.id = rt.tag_id
WHERE b.family_id = ? AND b.deleted_at IS NULL
  AND (? IS NULL OR b.bill_date BETWEEN ? AND ?)
  AND (? IS NULL OR b.category_l1 = ?)
  AND (? IS NULL OR b.payer_id = ?)
GROUP BY b.id
ORDER BY b.bill_date DESC, b.created_at DESC
LIMIT ? OFFSET ?;
```

### 5.2 统计 SQL（按一级分类）
```sql
SELECT category_l1, SUM(amount) AS total
FROM bill_record
WHERE family_id=? AND bill_date LIKE ?||'%' AND deleted_at IS NULL
GROUP BY category_l1 ORDER BY total DESC;
```
趋势：按月 GROUP BY `substr(bill_date,1,7)`。
成员对比：`GROUP BY payer_id`。
标签：JOIN `bill_record_tag` + `bill_tag`。

### 5.3 导入实现
1. 上传文件到 R2，建 `bill_import_job(status=parsing)`
2. Workers 解析（CSV 用原生；xlsx 用 `sheetjs`，注意 Workers 体积，可走 Queues 异步）
3. 字段映射 + 分类模糊匹配（基于内置分类名）
4. 校验：金额>0、日期合法、分类存在；失败行入 errorReport
5. 用户确认 → 批量 INSERT（事务）→ 去重（按 amount+billDate+category fingerprint）

export class NotifyDispatcher {
  async dispatch(event: NotifyEvent) {
    const members = await getTargetMembers(event);  // 全家 or 指定
    for (const m of members) {
      const pref = await getPref(m.id, event.type);
      if (pref.inapp)  await saveNotification(m, event);     // sys_notification
      if (pref.feishu) await sendFeishu(familyId, event);    // 飞书
      if (pref.webpush) await sendWebPush(m.id, event);       // Web Push
    }
  }
}

### 10.2 飞书自定义机器人

```ts
// workers/api/src/notify/FeishuBot.ts
import { createHmac } from 'crypto';

function sign(secret: string, timestamp: number): string {
  const stringToSign = `${timestamp}\n${secret}`;
  const hmac = createHmac('sha256', stringToSign);
  return hmac.update('').digest('base64');
}

export async function sendFeishu(cfg: FeishuConfig, title: string, content: string) {
  const timestamp = Math.floor(Date.now() / 1000);
  const body = {
    timestamp: String(timestamp),
    sign: cfg.secret ? sign(cfg.secret, timestamp) : undefined,
    msg_type: 'text',
    content: { text: `【家庭管家】${title}\n${content}` },
  };
  const res = await fetch(cfg.webhookUrl, {
    method: 'POST', headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  });
  const json = await res.json();
  if (json.code !== 0) throw new BizError('FEISHU_ERROR', json.msg);
}
```

> 富文本 / 卡片消息可用 `msg_type: 'interactive'`，V1 先用 text。
